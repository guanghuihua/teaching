\settime{00:30--03:00}
\begin{frame}[t]{先找回乘方的意思}\label{fr:review}
请独立算一算，再说说你是怎样算的。
\bigmath{2^2=\ ?\qquad 2^3=\ ?\qquad 3^2=\ ?}
\uncover<2->{\gap $2^2=2\times2=4$\par $2^3=2\times2\times2=8$\par $3^2=3\times3=9$}
\uncover<3->{\gap\key{$2^3$与$3^2$：位置换了，意思也换了。}}
\teachnote{给45秒独立计算，再请一人解释。若把2的3次方算成6，先回到三个2相乘。本页复习乘方，无需复数先修。}
\end{frame}

\settime{03:00--05:00}
\begin{frame}[t]{正方形怎样长大}\label{fr:square}
\begin{columns}[T,onlytextwidth]
\begin{column}{.43\textwidth}\centering\squarepicture\end{column}
\begin{column}{.54\textwidth}
每个小方格的面积为$1$。
\gap 边长从$3$增加到$4$，\par 一共多了几个小方格？
\uncover<2->{\gap $4^2=16$，原来有$3^2=9$。\par 新增$16-9=7$个。}
\uncover<3->{\gap 再长到边长$5$呢？\par $5^2=25$，又新增$9$个。}
\end{column}
\end{columns}
\teachnote{先想怎样补成4乘4，再点击补L形，避免一开始展示新增7。第三次点击前留20秒口算25减16。边长每次加1，面积并非每次加同一个数。}
\end{frame}

\settime{05:00--07:00}
\begin{frame}[t]{另一种长大：每轮翻倍}\label{fr:double}
开始只有$1$个点。每过一轮，每个点变成两个。
\bigmath{1\ \longrightarrow\ 2\ \longrightarrow\ 4\ \longrightarrow\ ?\ \longrightarrow\ ?\ \longrightarrow\ ?}
下一轮有几个？三轮以后呢？
\uncover<2->{\gap\centering $1\ \longrightarrow\ 2\ \longrightarrow\ 4\ \longrightarrow\ 8\ \longrightarrow\ 16\ \longrightarrow\ 32$\par}
\uncover<3->{\gap\raggedright\key{每轮乘$2$，是“翻倍”；每轮加$2$，不是翻倍。}}
\teachnote{开始是第0轮，经历1、2、3轮依次2、4、8。学生可画分叉图；检查第三轮是否错数为4。}
\end{frame}

\settime{07:00--09:00}
\begin{frame}[t]{两种规则，两个名字}\label{fr:names}
\begin{columns}[T,onlytextwidth]
\begin{column}{.47\textwidth}
\key{边长为$x$的正方形}
\bigmath{y={\color{warm}x}^{2}}
面积：$x$乘$x$。
\uncover<2->{\gap 变的是底数$x$，\par 指数$2$固定。\par 它是\key{幂函数}的一个例子。}
\end{column}
\begin{column}{.47\textwidth}
\key{从$1$开始，翻倍$x$轮}
\bigmath{y=2^{\color{warm}x}}
数量：$x$个$2$相乘。
\uncover<3->{\gap 底数$2$固定，\par 变的是指数$x$。\par 它是\key{指数函数}的一个例子。}
\end{column}
\end{columns}
\teachnote{翻倍模型先取正整数x，0轮数量为1可口头回顾。这里只给两个例子的名字，不推广一般底数、指数和定义域分类。非整数幂留待正式课程。}
\end{frame}

\settime{09:00--11:00}
\begin{frame}[t]{先猜：谁的结果更大}\label{fr:predict}
同样输入一个数，哪种规则得到的结果更大？
\bigmath{x^2\qquad\text{与}\qquad 2^x}
请先写下你的猜想，再算两个位置：
\gap \centering $x=3$时\hspace{22mm}$x=5$时\par
\uncover<2->{\gap\raggedright $3^2=9>8=2^3$\par $5^2=25<32=2^5$\gap\key{只算一个位置，还不能替所有位置作决定。}}
\teachnote{60秒先猜后算，两人交流；保留原猜想并修订，不立刻给“指数增长更快”的口号。}
\end{frame}

\settime{11:00--14:00}
\begin{frame}[t]{把比较做成一张数表}\label{fr:table}
先独立补表，再和同桌核对。可在草稿上连乘。
\gap\centering
\renewcommand{\arraystretch}{1.55}
\begin{tabular}{c|rrrrrrr}
$x$ &1&2&3&4&5&6&8\\\hline
$x^2$ &1&4&\uncover<2->{9}&\uncover<2->{16}&\uncover<2->{25}&\uncover<2->{36}&\uncover<2->{64}\\
$2^x$ &2&4&\uncover<2->{8}&\uncover<2->{16}&\uncover<2->{32}&\uncover<2->{64}&\uncover<2->{256}\\
\end{tabular}
\uncover<3->{\gap\raggedright 在表中圈出：\key{相等的位置}，以及\key{大小关系反过来的位置}。}
\teachnote{保留90秒以上补表。允许写2的8次方为64乘4。相等位置x为2、4；x3时平方大，x5、6、8时指数大。只讨论表中数据，不从有限点证明一般结论。}
\end{frame}

\settime{14:00--17:00}
\begin{frame}[t]{把数表画在同一张图上}\label{fr:points}
\renewcommand{\gap}{\par\vspace{2mm}}
\begin{columns}[T,onlytextwidth]
\begin{column}{.52\textwidth}\centering\growthplot\end{column}
\begin{column}{.45\textwidth}
绿圆点：$y=x^2$。
\gap 先找$(3,9)$和$(5,25)$。
\uncover<2->{\gap 橙方点：$y=2^x$。\par 再找$(3,8)$和$(5,32)$。}
\uncover<3->{\gap 同一个$x$，\par\key{比较两个点的高度。}\par $x=3$与$x=5$，\par 谁更高发生了变化。}
\end{column}
\end{columns}
\teachnote{先让学生定位再点击橙点。两组点共用相同坐标轴，仅画整数输入，不借小数指数增加负担。相等位置x2、4两标记重合；请学生指出表和图的对应，不从有限图窗推出所有大x。}
\end{frame}

\settime{17:00--19:00}
\begin{frame}[t]{一句话，能不能说得太满}\label{fr:counter}
小明说：\key{“$2^x$总是比$x^2$大。”}
\gap 你同意吗？请用刚才的一组数说明。
\uncover<2->{\bigmath{x=3:\qquad 2^3=8<9=3^2}}
\uncover<3->{\key{一个算对的反例，就能推翻“总是”。}\gap 今天看到：同一个$x$，两种规则有时相等，\par 有时一方大，有时另一方大。}
\teachnote{给20秒独立寻找反例。“不同意”必须附算式，不能只说图上看起来。不要求实数解集或证明指数最终超过平方。}
\end{frame}

\settime{19:00--21:00}
\begin{frame}[t]{每次加一，究竟加了多少}\label{fr:steps}
输入$x$每次增加$1$。比较下面两行的\key{增加量}。
\gap\centering
\renewcommand{\arraystretch}{1.4}
\begin{tabular}{c|rrrrr}
$x$ &1&2&3&4&5\\\hline
$x^2$ &1&4&9&16&25\\
$2^x$ &2&4&8&16&32\\
\end{tabular}
\uncover<2->{\gap\raggedright 平方这一行：依次增加$3,\ 5,\ 7,\ 9$。\par 翻倍这一行：依次增加$2,\ 4,\ 8,\ 16$。}
\uncover<3->{\gap\key{两行都不是每次加同一个数；第二行每次乘$2$。}}
\teachnote{请两组学生分别报相邻数之差，教师板书箭头。“增加量变大”不是指数函数独有，平方增加量也变大。本例关键是固定倍数。}
\end{frame}

\settime{21:00--23:00}
\begin{frame}[t]{输入翻倍，结果也翻倍吗}\label{fr:scale}
把输入从$3$变成$6$。请先算结果，再算倍数。
\begin{columns}[T,onlytextwidth]
\begin{column}{.47\textwidth}
\bigmath{x^2}
$9\ \longrightarrow\ ?$
\uncover<2->{\gap $6^2=36$\par $36\div9=4$\par 面积变为原来的\key{$4$倍}。}
\end{column}
\begin{column}{.47\textwidth}
\bigmath{2^x}
$8\ \longrightarrow\ ?$
\uncover<3->{\gap $2^6=64$\par $64\div8=8$\par 数量变为原来的\key{$8$倍}。}
\end{column}
\end{columns}
\teachnote{从3到6是乘2，不是加1。平方输入翻倍用边长翻倍解释；指数输出这里8倍，不能说输入翻倍就总8倍。后面用4到8核验。}
\end{frame}

\settime{23:00--25:00}
\begin{frame}[t]{现在更大，下一步也多加吗}\label{fr:value-step}
\begin{columns}[T,onlytextwidth]
\begin{column}{.47\textwidth}
\key{在$x=3$时，谁的值大？}
\bigmath{3^2\qquad2^3}
\uncover<2->{$9>8$\gap 平方这一项\key{现在更大}。}
\end{column}
\begin{column}{.47\textwidth}
\key{从$x=3$到$4$，谁多加？}
\gap 平方：$9\longrightarrow16$\par 翻倍：$8\longrightarrow16$
\uncover<3->{\gap 平方增加$7$，翻倍增加$8$。\par 翻倍这一项\key{这一步多加}。}
\end{column}
\end{columns}
\teachnote{分别用“值”和“增加量”作主语。这是同一个单位区间总增量，不说成某点瞬时速度，也不把更陡用于前面的散点。}
\end{frame}

\settime{25:00--28:00}
\begin{frame}[t]{自己做一次，不急着核对}\label{fr:practice}
在草稿上写出计算过程。
\gap 1. 输入$x=4$时，$x^2$与$2^x$的值相同吗？
\gap 2. 输入从$4$增加到$5$时，\par\hspace{5mm}两种结果各增加多少？
\gap 3. 输入从$4$翻倍到$8$时，\par\hspace{5mm}两种结果各变为原来的几倍？
\gap\color{muted}做完后，给同桌解释：“增加多少”和“变为几倍”怎样区分。
\teachnote{本页无答案，至少90秒独立书写，随后互讲。第3题迁移，可使用先前数表。收集原步骤再翻核对页，不把照抄当作独立完成。}
\end{frame}

\settime{28:00--30:00}
\begin{frame}[t]{核对时，找出差在了哪里}\label{fr:answers}
1. 输入$4$：$4^2=16$，$2^4=16$，\key{相同}。
\uncover<2->{\gap 2. 从$4$到$5$：\par\hspace{5mm}平方增加$25-16=9$；\par\hspace{5mm}翻倍增加$32-16=16$。}
\uncover<3->{\gap 3. 从$4$到$8$：\par\hspace{5mm}平方变为$64\div16=4$倍；\par\hspace{5mm}翻倍变为$256\div16=16$倍。}
\teachnote{首屏仅核对1，点击给2、3。做差回答增加多少，做除法回答几倍。第3题指数16倍与输入3到6的8倍不同，防止学生只记住一个倍数。}
\end{frame}

\settime{30:00--32:00}
\begin{frame}[t]{再小一点，看变化快慢}\label{fr:small-step}
仍看$y=x^2$，从$x=2$出发。
\gap 到$3$：横向增加$1$，纵向增加$9-4=5$。
\uncover<2->{\gap 到$2.1$：横向增加$0.1$，\par 纵向增加$2.1^2-4=4.41-4=0.41$。}
\uncover<3->{\gap 折算成“每增加$1$，平均增加多少”：\par 到$3$这一段：$5\div1=5$；\par 到$2.1$这一小段：$0.41\div0.1=4.1$。}
\teachnote{用路程除以时间类比折算每1单位，再用小数算式；不写差商通式。如小数除法卡住，提供4.1，重点辨认横向0.1与纵向0.41。}
\end{frame}

\settime{32:00--35:00}
\begin{frame}[t]{靠近一点，直线怎样转动}\label{fr:tangent}
\renewcommand{\gap}{\par\vspace{2mm}}
\begin{columns}[T,onlytextwidth]
\begin{column}{.51\textwidth}\centering\secantplot\end{column}
\begin{column}{.46\textwidth}
固定$P(2,4)$，让$Q$靠近$P$。
\uncover<2->{\gap $Q$在$x=3$：\par 平均每单位增加$5$。}
\uncover<3->{\gap $Q$在$x=2.1$：\par 平均每单位增加$4.1$。}
\uncover<4->{\gap $Q$越来越靠近，直线趋近\key{切线}。\par 它的倾斜程度，描述\par\key{此刻的变化快慢}。}
\end{column}
\end{columns}
\teachnote{手动四个状态，先固定P，再出现远Q、近Q、切线。可口述Q在2.01时平均4.01，感受靠近4。本例切线斜率4；几条线是示意而非严格证明。最多口述“研究这一点的变化快慢，后来叫求导数”，不给新符号或求导规则。切线不以只碰曲线一次作一般定义。若前面练习用时长，此页缩到1分钟，保留末5分钟AI。}
\end{frame}

\settime{35:00--37:00}
\begin{frame}[t]{用自己的话说明白}\label{fr:retell}
两人一组，轮流说一句，并用算式支持。
\gap 1. $x^2$与$2^x$，变的是哪个位置？
\uncover<2->{\gap 2. 为什么不能说“$2^x$总是更大”？\par{\color{muted}可用$x=3$的一组数作证。}}
\uncover<3->{\gap 3. “现在的值大”和“下一步多加”，一样吗？\par{\color{muted}可用从$x=3$到$4$的两行数说明。}}
\teachnote{每人各说30秒。只念名称不算完成；听变量位置、反例9与8、同区间增量7与8。第2、3层是追问，不替学生给完整解释。}
\end{frame}

\settime{37:00--40:00}
\begin{frame}[t]{最后两问，检查是否真的懂}\label{fr:exit}
\renewcommand{\gap}{\par\vspace{2mm}}
已知$x=5$时两种结果为$25$和$32$；\par $x=6$时两种结果为$36$和$64$。
\gap 1. 从$5$到$6$，哪一种多增加？请写减法。
\gap 2. “两种规则都是每次加同一个数。”对吗？\par\hspace{5mm}请用两次相邻变化说明。
\uncover<2->{\gap 1. 平方增加$11$；翻倍增加$32$。}
\uncover<3->{\gap 2. 不对。例如平方前两次增加$3,5$；\par\hspace{5mm}翻倍前两次增加$2,4$，都不同。}
\teachnote{留45秒独立小纸条或草稿，核对后回应错误。题1错往往是直接比值而没做差；题2只说不对但没给两次增量，证据不足。按算式与解释反馈，不新增正式考试分数。}
\end{frame}

\settime{40:00--42:00}
\begin{frame}[t]{留给AI的一个新问题}\label{fr:ai-question}
如果把“平方”换成“立方”呢？
\bigmath{x^3\qquad\text{与}\qquad2^x}
